cmake_minimum_required(VERSION 3.20)
project(XAIArchitectureResearch LANGUAGES CXX)

include(CTest)

find_path(GMP_INCLUDE_DIR gmpxx.h REQUIRED)
find_library(GMPXX_LIBRARY NAMES gmpxx REQUIRED)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)

add_executable(xai_pillar_tests TESTS/cpp/pillar_tests.cpp)
target_compile_features(xai_pillar_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(xai_pillar_tests PRIVATE /W4 /WX)
else()
  target_compile_options(xai_pillar_tests PRIVATE -Wall -Wextra -Werror)
endif()

add_library(xai_contracts src/contracts.cpp)
target_include_directories(xai_contracts PUBLIC include)
target_compile_features(xai_contracts PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(xai_contracts PRIVATE /W4 /WX)
else()
  target_compile_options(xai_contracts PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(xai_contract_tests TESTS/cpp/contracts_tests.cpp)
target_link_libraries(xai_contract_tests PRIVATE xai_contracts)
target_compile_features(xai_contract_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(xai_contract_tests PRIVATE /W4 /WX)
else()
  target_compile_options(xai_contract_tests PRIVATE -Wall -Wextra -Werror)
endif()

add_library(xai_wmc src/wmc.cpp)
target_include_directories(xai_wmc PUBLIC include ${GMP_INCLUDE_DIR})
target_compile_features(xai_wmc PUBLIC cxx_std_20)
target_link_libraries(xai_wmc PUBLIC ${GMPXX_LIBRARY} ${GMP_LIBRARY})
if(MSVC)
  target_compile_options(xai_wmc PRIVATE /W4 /WX /O2)
else()
  target_compile_options(xai_wmc PRIVATE -Wall -Wextra -Werror -O3)
endif()

add_executable(xai_wmc_solver src/main.cpp)
target_link_libraries(xai_wmc_solver PRIVATE xai_wmc)
target_compile_features(xai_wmc_solver PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(xai_wmc_solver PRIVATE /W4 /WX /O2)
else()
  target_compile_options(xai_wmc_solver PRIVATE -Wall -Wextra -Werror -O3)
endif()

add_executable(xai_wmc_tests TESTS/cpp/wmc_tests.cpp)
target_link_libraries(xai_wmc_tests PRIVATE xai_wmc)
target_compile_features(xai_wmc_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(xai_wmc_tests PRIVATE /W4 /WX /O2)
else()
  target_compile_options(xai_wmc_tests PRIVATE -Wall -Wextra -Werror -O3)
endif()

if(BUILD_TESTING)
  add_test(NAME xai_pillar_tests COMMAND xai_pillar_tests)
  add_test(NAME xai_contract_tests COMMAND xai_contract_tests)
  add_test(NAME xai_wmc_tests COMMAND xai_wmc_tests)
endif()
